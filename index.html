Kode Google Apps Script (Backend)

Aplikasi ini terhubung langsung ke Google Apps Script Anda. Pastikan kode berikut telah terpasang di editor script.google.com milik sekolah: 

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    
    // 1. Link Folder Penyimpanan
    var docTemplateId = "1GrTCFV0pdszmc1YenSbQLJ_5XpRAWjFSm2MIHKPhBlE"; // Template Doc
    var docFolderId   = "1dJVlX6xYOHUc2H5TsP9A7Z_rgSF9xiEY"; // Folder Formulir Pendaftar
    var fileFolderId  = "1-LGTdy1XyHR0gefvSyGqEEwIu7XrgWCC"; // Folder Berkas Upload

    var targetDocFolder  = DriveApp.getFolderById(docFolderId);
    var targetFileFolder = DriveApp.getFolderById(fileFolderId);

    // 2. Upload Berkas (Pas Foto, KK, Akte, Bukti Transfer)
    var uploadedUrls = {};
    var fileKeys = ['foto', 'kk', 'akte', 'transfer'];
    
    fileKeys.forEach(function(key) {
      if (data.files && data.files[key]) {
        var fileInfo = data.files[key];
        var bytes = Utilities.base64Decode(fileInfo.data);
        var blob = Utilities.newBlob(bytes, fileInfo.type, data.namaLengkap + "_" + key.toUpperCase() + "_" + fileInfo.name);
        var savedFile = targetFileFolder.createFile(blob);
        savedFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        uploadedUrls[key] = savedFile.getUrl();
      }
    });

    // 3. Duplikasi Template Doc & Isi Data Pendaftar
    var templateFile = DriveApp.getFileById(docTemplateId);
    var newDocFile = templateFile.makeCopy(data.namaLengkap + " - Formulir SPMB", targetDocFolder);
    var doc = DocumentApp.openById(newDocFile.getId());
    var body = doc.getBody();

    // Replace Placeholders dalam Dokumen
    body.replaceText("{{NAMA}}", data.namaLengkap || "");
    body.replaceText("{{NISN}}", data.nisn || "");
    body.replaceText("{{NIK}}", data.nik || "");
    body.replaceText("{{TTL}}", (data.tempatLahir + ", " + data.tanggalLahir) || "");
    body.replaceText("{{JK}}", data.jenisKelamin || "");
    body.replaceText("{{AGAMA}}", data.agama || "");
    body.replaceText("{{ALAMAT}}", data.alamatSiswa || "");
    body.replaceText("{{SEKOLAH}}", data.sekolahAsal || "");
    body.replaceText("{{HP_SISWA}}", data.noHpSiswa || "");
    body.replaceText("{{AYAH}}", data.namaAyah || "");
    body.replaceText("{{PEKERJAAN_AYAH}}", data.pekerjaanAyah || "");
    body.replaceText("{{IBU}}", data.namaIbu || "");
    body.replaceText("{{PEKERJAAN_IBU}}", data.pekerjaanIbu || "");
    body.replaceText("{{HP_ORTU}}", data.noHpOrtu || "");
    body.replaceText("{{TANGGAL_DAFTAR}}", new Date().toLocaleDateString('id-ID'));

    doc.saveAndClose();

    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      message: "Data & Berkas Berhasil Disimpan",
      docUrl: newDocFile.getUrl(),
      regId: "REG-" + Math.floor(1000 + Math.random() * 9000)
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
