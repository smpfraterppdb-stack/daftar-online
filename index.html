Langkah Mudah Mengaktifkan Backend Google Drive:

  1. Buka akun Google Drive sekolah, buat script baru di script.google.com.
  2. Salin (copy) seluruh kode Google Apps Script di bawah ini dan tempelkan ke editor script:
// === KODE GOOGLE APPS SCRIPT SMP FRATER MAMASA ===
const TEMPLATE_DOC_ID = '1GrTCFV0pdszmc1YenSbQLJ_5XpRAWjFSm2MIHKPhBlE';
const TARGET_DOC_FOLDER_ID = '1dJVlX6xYOHUc2H5TsP9A7Z_rgSF9xiEY';
const TARGET_UPLOAD_FOLDER_ID = '1-LGTdy1XyHR0gefvSyGqEEwIu7XrgWCC';

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    
    // 1. Buat Dokumen Baru dari Template dengan Nama Siswa
    const templateFile = DriveApp.getFileById(TEMPLATE_DOC_ID);
    const targetFolder = DriveApp.getFolderById(TARGET_DOC_FOLDER_ID);
    const newDocFile = templateFile.makeCopy('Formulir - ' + data.namaLengkap, targetFolder);
    const doc = DocumentApp.openById(newDocFile.getId());
    const body = doc.getBody();
    
    // Replace Placeholder di Google Doc Template
    body.replaceText('{{NAMA_LENGKAP}}', data.namaLengkap || '');
    body.replaceText('{{NISN}}', data.nisn || '');
    body.replaceText('{{NIK}}', data.nik || '');
    body.replaceText('{{JENIS_KELAMIN}}', data.jenisKelamin || '');
    body.replaceText('{{TEMPAT_LAHIR}}', data.tempatLahir || '');
    body.replaceText('{{TANGGAL_LAHIR}}', data.tanggalLahir || '');
    body.replaceText('{{AGAMA}}', data.agama || '');
    body.replaceText('{{SEKOLAH_ASAL}}', data.sekolahAsal || '');
    body.replaceText('{{ALAMAT}}', data.alamat || '');
    body.replaceText('{{NAMA_AYAH}}', data.namaAyah || '');
    body.replaceText('{{PEKERJAAN_AYAH}}', data.pekerjaanAyah || '');
    body.replaceText('{{NAMA_IBU}}', data.namaIbu || '');
    body.replaceText('{{PEKERJAAN_IBU}}', data.pekerjaanIbu || '');
    body.replaceText('{{NO_HP}}', data.noHp || '');
    body.replaceText('{{EMAIL}}', data.email || '');
    body.replaceText('{{NO_REG}}', data.noReg || '');
    
    doc.saveAndClose();

    // 2. Unggah Berkas ke Folder Upload
    const uploadFolder = DriveApp.getFolderById(TARGET_UPLOAD_FOLDER_ID);
    
    function saveFile(base64Data, originalName, prefix) {
      if(!base64Data) return null;
      const splitData = base64Data.split(',');
      const contentType = splitData[0].match(/:(.*?);/)[1];
      const bytes = Utilities.base64Decode(splitData[1]);
      const blob = Utilities.newBlob(bytes, contentType, prefix + '_' + data.namaLengkap + '_' + originalName);
      return uploadFolder.createFile(blob).getUrl();
    }

    const fotoUrl = saveFile(data.fileFoto, 'Foto.jpg', 'Pasfoto');
    const kkUrl = saveFile(data.fileKk, 'KK.pdf', 'KK');
    const aktaUrl = saveFile(data.fileAkta, 'Akta.pdf', 'Akta');
    const transferUrl = saveFile(data.fileTransfer, 'Transfer.jpg', 'BuktiTransfer');

    return ContentService.createTextOutput(JSON.stringify({
      status: 'success',
      message: 'Data dan Berkas Berhasil Disimpan ke Google Drive!',
      docUrl: newDocFile.getUrl(),
      noReg: data.noReg
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      status: 'error',
      message: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
  1. Klik tombol Deploy (Terapkan) > New Deployment (Penerapan Baru).

  3. Setel Execute as: Me (SAYA) dan Who has access: Anyone (SIAPA SAJA).
